cmake_minimum_required(VERSION 3.13)
project(imx6ull_rtsp_mjpeg_streaming LANGUAGES CXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Full targets require Linux V4L2/POSIX headers. Use Ubuntu; Windows portable checks are not a Linux build.")
endif()
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(CMAKE_CROSSCOMPILING)
    option(BUILD_TESTING "Build tests (never run ARM tests on the host)" OFF)
    option(LP2_BUILD_TOOLS "Build native JPEG checker and RTP receiver" OFF)
else()
    option(BUILD_TESTING "Build offline protocol and ownership tests" ON)
    option(LP2_BUILD_TOOLS "Build native JPEG checker and RTP receiver" ON)
endif()
include(CTest)
set(LP2_JPEG_FIXTURE "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/synthetic_720p.jpg"
    CACHE FILEPATH "Non-private baseline 4:2:0 JPEG fixture")
if(BUILD_TESTING AND NOT EXISTS "${LP2_JPEG_FIXTURE}")
    message(FATAL_ERROR "JPEG fixture missing; do not skip the tests")
endif()
add_subdirectory(m5_rtsp_server)
if(LP2_BUILD_TOOLS)
    add_subdirectory(m4_rtp_jpeg)
    add_subdirectory(m4_rtp_jpeg_check)
endif()
